# --- REVISIONIST POWER AGGRESSION (ERA 3 & 4) ---

	allowed = {
		OR = {
			original_tag = SOV
			original_tag = CHI
			original_tag = PER
			original_tag = NKO
		}
	}
	enable = {
		OR = {
			has_global_flag = tw2030_era_3_active
			has_global_flag = tw2030_era_4_active
		}
		is_ai = yes
	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = avoid_starting_wars
		value = -100
	}
}
