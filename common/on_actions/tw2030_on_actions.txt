# Total War 2030 - On Actions

on_actions = {
	# on_monthly runs once per country; the global flag is set by the first country
	# of the month so the pulse (and its every_country loop) runs only once.
	on_monthly = {
		effect = {
			if = {
				limit = {
					NOT = { has_global_flag = tw2030_monthly_done }
				}
				set_global_flag = {
					flag = tw2030_monthly_done
					days = 25
				}
				tw2030_escalation_monthly_pulse = yes
			}
		}
	}
}
